        function switchTab(tabName) {
            currentTab = tabName;
            
            ['qa', 'library', 'cards', 'quiz'].forEach(t => {
                const sec = document.getElementById(`tab-${t}`);
                if (sec) {
                    sec.classList.add('hidden');
                    sec.classList.remove('flex');
                }
                
                const navBtn = document.getElementById(`nav-${t}`);
                if (navBtn) {
                    navBtn.classList.remove('text-pink-700', 'bg-white', 'shadow-sm', 'border-pink-200');
                    navBtn.classList.add('text-slate-600');
                }
            });

            const activeSection = document.getElementById(`tab-${tabName}`);
            if (activeSection) {
                activeSection.classList.remove('hidden');
                activeSection.classList.add('flex');
            }

            const activeBtn = document.getElementById(`nav-${tabName}`);
            if (activeBtn) {
                activeBtn.classList.add('text-pink-700', 'bg-white', 'shadow-sm', 'border-pink-200');
                activeBtn.classList.remove('text-slate-600');
            }
        }

        async function handleChatSubmit(event) {
            event.preventDefault();
            const inputEl = document.getElementById('chat-input');
            const message = inputEl.value.trim();
            if (!message) return;

            const starterChips = document.getElementById('starter-chips');
            if (starterChips) starterChips.style.display = 'none';

            // Display User Message
            appendMessage('user', message);
            inputEl.value = '';

            // Check if user is requesting a textbook
            const matchedBook = checkTextbookRequest(message);

            if (matchedBook) {
                const loadingId = appendLoadingMessage();
                setTimeout(() => {
                    removeMessage(loadingId);
                    appendBookResponseCard(matchedBook);
                }, 600);
                return;
            }

            // Check if user asked a standard curriculum question offline DB first
            const localMatch = checkOfflineKnowledge(message);
            if (localMatch) {
                const loadingId = appendLoadingMessage();
                setTimeout(() => {
                    removeMessage(loadingId);
                    appendMessage('assistant', localMatch);
                }, 500);
                return;
            }

            // Otherwise, attempt Gemini API with fallback
            const loadingId = appendLoadingMessage();
            const gradeName = document.getElementById('user-grade-selector').options[document.getElementById('user-grade-selector').selectedIndex].text;
            
            const systemPrompt = `أنت "مساعد أم عامر الذكي" المخصص لثانوية أم عامر الأنصارية بنات - دولة الكويت (${gradeName}).
قانون الإجابة المباشر:
1. أجب باللغة العربية بأسلوب راقٍ، واضح، ومباشر وبدون مقدمات طويلة.
2. استخدم صيغة (س و ج):
   **س:** [صياغة السؤال]
   **ج:** [الإجابة النموذجية المعتمدة في الكويت]`;

            try {
                if (!apiKey) throw new Error("No API Key Provided");
                const responseText = await callGeminiAPI(message, systemPrompt);
                removeMessage(loadingId);
                appendMessage('assistant', responseText);
            } catch (err) {
                removeMessage(loadingId);
                // Smart Fallback Answer
                const fallbackResponse = generateSmartFallbackAnswer(message, gradeName);
                appendMessage('assistant', fallbackResponse);
            }
        }

        function checkOfflineKnowledge(query) {
            const lower = query.toLowerCase();
            for (let item of offlineAnswersDB) {
                if (item.keywords.some(k => lower.includes(k))) {
                    return item.response;
                }
            }
            return null;
        }

        function checkTextbookRequest(text) {
            const lower = text.toLowerCase();
            const keywords = ['كتاب', 'الكتب', 'اريد', 'أريد', 'تحميل', 'تنزيل', 'محمل', 'pdf', 'كراسة', 'بنك'];
            const containsKeyword = keywords.some(k => lower.includes(k));

            if (!containsKeyword) return null;

            let allBooks = [];
            Object.keys(booksDatabase).forEach(k => {
                allBooks = allBooks.concat(booksDatabase[k]);
            });

            if (lower.includes('11') || lower.includes('حادي عشر')) {
                if (lower.includes('رياضيات')) return allBooks.find(b => b.id === 'math-11-sci-t1');
                if (lower.includes('فيزياء')) return allBooks.find(b => b.id === 'physics-11-sci-t1');
                if (lower.includes('كيمياء')) return allBooks.find(b => b.id === 'chem-11-sci-t1');
                if (lower.includes('أحياء') || lower.includes('احياء')) return allBooks.find(b => b.id === 'bio-11-sci-t1');
                if (lower.includes('عربي') || lower.includes('لغة عربية')) return allBooks.find(b => b.id === 'arabic-11-t1');
                if (lower.includes('انجلش') || lower.includes('انجليزي') || lower.includes('english')) return allBooks.find(b => b.id === 'english-11-t1');
                return allBooks.find(b => b.id === 'math-11-sci-t1');
            }

            if (lower.includes('12') || lower.includes('ثاني عشر')) {
                if (lower.includes('رياضيات')) return allBooks.find(b => b.id === 'math-12-sci-t1');
                if (lower.includes('فيزياء')) return allBooks.find(b => b.id === 'physics-12-sci-t1');
                if (lower.includes('تاريخ')) return allBooks.find(b => b.id === 'history-12-lit-t1');
            }

            if (lower.includes('10') || lower.includes('عاشر')) {
                return allBooks.find(b => b.id === 'math-10-t1');
            }

            if (lower.includes('رياضيات')) return allBooks.find(b => b.id === 'math-11-sci-t1');
            if (lower.includes('فيزياء')) return allBooks.find(b => b.id === 'physics-11-sci-t1');
            if (lower.includes('كيمياء')) return allBooks.find(b => b.id === 'chem-11-sci-t1');

            return null;
        }

        async function callGeminiAPI(prompt, systemInstructionText) {
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

            const payload = {
                contents: [{ role: "user", parts: [{ text: prompt }] }],
                systemInstruction: { parts: [{ text: systemInstructionText }] }
            };

            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!response.ok) throw new Error('API Request Failed');

            const result = await response.json();
            const text = result.candidates?.[0]?.content?.parts?.[0]?.text;
            if (text) return text;
            throw new Error('Empty response');
        }

        async function generateAIQuiz() {
            const subject = document.getElementById('quiz-subject').value;
            const topic = document.getElementById('quiz-topic').value;
            const count = parseInt(document.getElementById('quiz-count').value) || 3;

            const btn = document.getElementById('generate-quiz-btn');
            btn.disabled = true;
            btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i> <span>جاري توليد الأسئلة...</span>`;

            // Try API generation if key exists
            if (apiKey) {
                try {
                    const prompt = `قم بإنشاء اختبار اختيار من متعدد مكون من ${count} أسئلة لمادة (${subject}) منهج الكويت. ${topic ? 'الموضوع: ' + topic : ''}.`;
                    const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

                    const payload = {
                        contents: [{ parts: [{ text: prompt }] }],
                        generationConfig: {
                            responseMimeType: "application/json",
                            responseSchema: {
                                type: "ARRAY",
                                items: {
                                    type: "OBJECT",
                                    properties: {
                                        question: { type: "STRING" },
                                        options: { type: "ARRAY", items: { type: "STRING" } },
                                        correctIndex: { type: "INTEGER" },
                                        explanation: { type: "STRING" }
                                    },
                                    required: ["question", "options", "correctIndex", "explanation"]
                                }
                            }
                        }
                    };

                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (response.ok) {
                        const result = await response.json();
                        const jsonText = result.candidates?.[0]?.content?.parts?.[0]?.text;
                        if (jsonText) {
                            generatedQuestions = JSON.parse(jsonText);
                            userAnswers = {};
                            renderQuizUI();
                            btn.disabled = false;
                            btn.innerHTML = `<i class="fa-solid fa-wand-magic-sparkles"></i> <span>إنشاء الأسئلة الآن</span>`;
                            return;
                        }
                    }
                } catch (e) {
                    console.log('Using local quiz fallback generator');
                }
            }

            // Reliable Local Generator Fallback
            setTimeout(() => {
                const subjectPool = localQuizDatabase[subject] || localQuizDatabase['الرياضيات'];
                generatedQuestions = subjectPool.slice(0, count);
                userAnswers = {};
                renderQuizUI();
                btn.disabled = false;
                btn.innerHTML = `<i class="fa-solid fa-wand-magic-sparkles"></i> <span>إنشاء الأسئلة الآن</span>`;
            }, 500);
        }

        function evaluateQuiz() {
            let score = 0;
            generatedQuestions.forEach((q, qIdx) => {
                const selected = userAnswers[qIdx];
                const expDiv = document.getElementById(`explanation-${qIdx}`);
                expDiv.classList.remove('hidden');

                if (selected === q.correctIndex) {
                    score++;
                    expDiv.innerHTML = `✓ إجابة صحيحة: ${q.explanation}`;
                    expDiv.className = "p-3 rounded-xl bg-emerald-100 border border-emerald-300 text-emerald-800 text-xs font-bold";
                } else {
                    const correctText = q.options[q.correctIndex];
                    expDiv.innerHTML = `✗ الإجابة الصحيحة هي (${correctText}) — ${q.explanation}`;
                    expDiv.className = "p-3 rounded-xl bg-rose-100 border border-rose-300 text-rose-800 text-xs font-bold";
                }
            });

            const scoreBox = document.createElement('div');
            scoreBox.className = "p-4 rounded-2xl bg-gradient-to-r from-pink-500 to-rose-500 text-white text-center font-extrabold text-base mb-4 shadow-md";
            scoreBox.innerText = `النتيجة المباشرة: ${score} من أصل ${generatedQuestions.length}`;
            
            const container = document.getElementById('quiz-container');
            container.insertBefore(scoreBox, container.firstChild);
            document.getElementById('submit-quiz-btn').style.display = 'none';
        }
    </script>
